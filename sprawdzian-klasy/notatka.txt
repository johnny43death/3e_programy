class Prostopadloscian {
private:
    int a, b, h;

        cout << "Wartosc boku b podstawy: ";
        cin >> b_i;
        cout << "Wartosc wysokosci: ";
        cin >> h_i;
        a = a_i;
        b = b_i;
        h = h_i;
    };
    void pole() {
        cout << "Pole powierzchni calkowitej prostopadloscianu wynosi: " << ((2 * a * b) + (2 * a * h) + (2 * b * h));

        cout << "Objetosc prostopadloscianu wynosi " << (a * b * h);
    };
    void krawedzie() {
        cout << "Bok a podstawy: " << a << " bok b podstawy: " << b << " wysokosc prostopadloscianu: " << h;
    };
};


hermetyzacja i dziedziczenie
hermetyzacja, wiadomo - w public jest wszystko dostępne, 
dla private to trzeba robić gettery i settery robić żeby mieć do tego dostęp i aby to zmieniać

nie dziedziczy się tam tego co w private, za to co w protected się dziedziczy
można tez dziedziczyć pisząc : private Klasa { lub : protected Klasa { ale to działa jakoś inaczej w kwestii
tego co jest dziedziczone

Są 3 typy dziedziczenia: publiczne (public), chronione (protected) i prywatne (private). Różnią się dostępem do odziedziczonych składowych w klasie pochodnej:

Dziedziczenie publiczne jest najczęściej stosowane. Składowe publiczne klasy bazowej są odziedziczone jako publiczne, a składowe chronione jako chronione.
W dziedziczeniu chronionym składowe publiczne są dziedziczone jako chronione, a składowe chronione jako chronione.
Dziedziczenie prywatne jest domyślne (gdy nie jest podany typ dziedziczenia). Składowe publiczne są dziedziczone jako prywatne, a chronione jako prywatne.

We wszystkich sposobach dziedziczenia, składowe prywatne klasy bazowej są dziedziczone jako prywatne, ale klasa pochodna nie ma do nich dostępu. Aby klasa pochodna miała dostęp do prywatnych składowych klasy bazowej, musi być zaprzyjaźniona z klasą bazową.

funkcje i klasy zaprzyjaźnione
Zdarza się, że jakaś klasa/funkcja potrzebuje dostępu do prywatnych lub chronionych składowych klasy. 
Aby zezwolić na dostęp, używa się deklaracji przyjaźni miedzy klasą a inną klasą lub funkcją.

Deklaracje przyjaźni umieszcza się w klasie, 
która ma udostępniać swoje składowe wybranym klasom i/lub funkcjom.


class Klasa
{
    int moj_tajny_int;
   
    friend class ZaprzyjaznionaKlasa; //friend <deklaracja klasy>
    friend void ZaprzyjaznionaFunkcja( Klasa & ); //friend <deklaracja funkcji>
};

class ZaprzyjaznionaKlasa
{
public:
    void metoda( Klasa & k )
    {
        k.moj_tajny_int = 0; //Dostęp do prywatnej składowej innej klasy
    }
};

void ZaprzyjaznionaFunkcja( Klasa & k )
{
    k.moj_tajny_int = 0;
}

int main()
{
    Klasa k;
   
    ZaprzyjaznionaKlasa zk;
    zk.metoda( k );
    ZaprzyjaznionaFunkcja( k );
}


Funkcja zaprzyjaźniona może być zdefiniowana wewnątrz klasy:

class Klasa
{
    friend void funkcja() //funkcja, a nie metoda
    {
    }

szablony funkcji i klas


#include <iostream>
using namespace std;
/*
specjalizowane szablony klas również ma wyspecjalizowaną wersję dla danego typu danych lub dla danych typów danych (jeśli klasa ma wiele parametrów)

*/
template <class T>
class Kolo {
public:
    T r;
    T Obwod() {
        return 3 * round(r) * 2;
    }
};


template <> class
Kolo<float> {
public:
    float r;
    float Obwod() {
        return 3.14 * r * 2;
    }
};


template <> class
Kolo<double> {
public:
    double r;
    double Obwod() {
        return 3.141592653589793238462643 * r * 2;
    }
};



int main()
{
    Kolo<int> kolo1{ 4 };
    Kolo<float> kolo2{ 3.6 };
    Kolo<double> kolo3{ 7 };
    cout << "1: " << kolo1.Obwod() << endl;
    cout << "2: " << kolo2.Obwod() << endl;
    cout << "3: " << kolo3.Obwod() << endl;
}



#include <iostream>
using namespace std;

/*specjalizowane szablony funkcji charakteryzują się tym że mają jedną lub więcej specjalnych wersji używanych przy danym typie danych
taka wersja jest funkcją specjalizowana
implementacja jej powinna się różnić od implementacji szablonu

*/
template <typename T>
T srednia(T input_1, T input_2) {
	cout << "\ndefault template" << endl;
	return (input_1 + input_2) / 2;
}
template <>
float srednia(float input_1, float input_2) {
	cout << "\nfloat" << endl;
	return (round(input_1) + round(input_2)) / 2;
}
template <>
double srednia(double input_1, double input_2) {
	cout << "\ndouble" << endl;
	return (round(input_1) + round(input_2)) / 2;
}

int main()
{
	//kompilator sprawdza najpierw czy istnieją szablony specjalizowane dla danego typu danych a jeśli nie istnieją używa domyślnego szablonu
	cout << srednia(3, 5) << endl;
	cout << srednia(3.8, 2.21) << endl;
	cout << srednia<double>(3.8, 2.21) << endl;
	cout << srednia<float>(3.8, 2.21) << endl;
}

#include <iostream>
#include <string>
#include <cstdlib>
#include <cmath>
#include <sstream>

using namespace std;

struct Wyjatek {
    virtual string komunikat() = 0;
};

class Format : public Wyjatek {
    string sOcena;
public:
    Format(string sOcena) : sOcena(sOcena) {};
    string komunikat() {
        stringstream sTemp;
        sTemp << "UWAGA BLAD! Format danej wejsciowej " << sOcena << " nie odpowiada liczbie calkowitej" << endl;
        return sTemp.str();
    }
};
class Zakres : public Wyjatek {
    int ocena;
public:
    //do Zakresu przypisujemy wartość zmiennej ocena(ocena)
    Zakres(int ocena) : ocena(ocena) {};
    string komunikat() {
        stringstream sTemp;
        sTemp << "UWAGA BLAD! Wartosc danej wejsciowej " << ocena << " nie miesci sie w zakresie <1,6>" << endl;
        return sTemp.str();
    }
};

bool wejscieFormat(string sDana) {
    for (int i = 0; i < sDana.length(); i++) {
        if (isdigit(sDana[i]) == false) {
            return false;
        }
    }
    return true;
}

bool wejscieZakres(int iDana) {
    if ((iDana >= 1) && (iDana <= 6)) {
        return true;
    } else {
        return false;
    }
}

int main()
{
    try {
        string sOcena;
        cin >> sOcena;
        if(wejscieFormat(sOcena) == false) {
            throw Format(sOcena);
        }

        int ocena = stoi(sOcena);

        if (wejscieZakres(ocena) == false) {
            throw Zakres(ocena);
        }

        if (ocena == 1) {
            cout << "uczen nie otrzymal promocji";
        }
        else {
            cout << "uczen otrzymuje promocje";
        }
    }
    
    catch (Wyjatek& wyjatek) {
        cout << wyjatek.komunikat() << endl;
    }   
}
